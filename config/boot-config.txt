# Démarrer la sortie HDMI même si la TV est éteinte ou sur une autre source
# quand le Pi s'allume (sinon : écran noir en allumant la TV après le Pi).
hdmi_force_hotplug=1

# Mode HDMI « TV » : le son passe par le câble HDMI, pas par la prise jack.
hdmi_drive=2

# Supprime les bandes noires autour de l'image. Si l'image déborde de l'écran,
# réglez d'abord la TV en mode « Just Scan / Plein écran / 1:1 », puis
# seulement en dernier recours, passez cette valeur à 0.
disable_overscan=1

# Mémoire réservée au GPU : 256 Mo est la valeur conseillée par RetroPie
# pour le Pi 3 (N64, PS1 en résolution plus élevée).
gpu_mem=256
